--- 파일 1: apply-manual-zip.bat ---
@echo off
cd /d "%~dp0"
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
 "$z=$args[0]; if(-not $z){$d=[Environment]::GetFolderPath('UserProfile')+'\Downloads'; $z=(Get-ChildItem $d -Filter 'automatic_stock_trading*.zip'|Sort LastWriteTime -Desc|Select -First 1).FullName}; if(-not $z){Write-Host 'ZIP 없음: GitHub에서 Code-Download ZIP'; pause; exit 1}; $r=(Get-Location).Path; $t=$env:TEMP+'\aidi'+[guid]::NewGuid().ToString('n'); $cp=Join-Path $r 'backend\data\credentials.json'; $bk=Join-Path $t 'c.json'; if(Test-Path $cp){New-Item -ItemType Directory -Force -Path $t|Out-Null; Copy-Item $cp $bk -Force}; Expand-Archive -LiteralPath $z -DestinationPath $t -Force; $s=Join-Path $t 'automatic_stock_trading-main'; if(-not(Test-Path $s)){$s=(Get-ChildItem $t -Directory|Select -First 1).FullName}; robocopy $s $r /E /XD backend\.venv backend\data .git node_modules /NFL /NDL /NJH /NJS; if(Test-Path $bk){Copy-Item $bk $cp -Force}; Write-Host Done. run.bat 실행; pause" ^
 "%~1"
